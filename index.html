Google Sheets — Istruzioni scrittura

La lettura dal foglio funziona automaticamente. Per salvare e eliminare movimenti dal cloud, segui questi passaggi:

  1. Apri il tuo Google Sheet dei Movimenti
  2. Vai su Estensioni → Apps Script
  3. Incolla il codice qui sotto e salva
  4. Clicca Deploy → New deployment → Web App
  5. Imposta "Execute as: Me" e "Who has access: Anyone"
  6. Copia l'URL del deploy
  7. Incollalo nel campo URL nella barra di sincronizzazione del Gestionale
Mostra codice Apps Script 
const SHEET_ID = '1NILR4iJdoGProvayhCLYSGuzkd-7ouyJOD4HwFTvFPA';
const SHEET_NAME = 'Movimenti';
const HEADERS = ['ID','Tipo','Data','Prodotto/Varietà','Cliente/Fornitore','Totale €','Fatturato','N° Fattura','Kg/Qtà','Extra JSON','DELETED'];


function doPost(e) {
  try {
    let data;
    if (e.postData.type === 'application/x-www-form-urlencoded' || e.parameter.data) {
      data = JSON.parse(e.parameter.data);
    } else {
      data = JSON.parse(e.postData.contents);
    }
    const ss = SpreadsheetApp.openById(SHEET_ID);


    // MOVIMENTI
    const sh = ss.getSheetByName(SHEET_NAME) || ss.insertSheet(SHEET_NAME);
    if (sh.getLastRow() === 0) sh.appendRow(HEADERS);
    if (data.action === 'append') { sh.appendRow(data.row); }
    if (data.action === 'delete') {
      const vals = sh.getDataRange().getValues();
      for (let i = 1; i < vals.length; i++) {
        if (String(vals[i][0]) === String(data.id)) { sh.getRange(i+1,11).setValue('YES'); break; }
      }
    }
    return ContentService.createTextOutput(JSON.stringify({ok:true})).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ok:false,error:err.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  return ContentService.createTextOutput(JSON.stringify({ok:true})).setMimeType(ContentService.MimeType.JSON);
}
